  Person (abstract)
    ├── Applicant (email, address, registrationDate)
    └── Admin (role, department)

MODELS:
  Condominium (condoId, name, location, floors, units)
  Unit (unitId, condoId, floor, number, area, price, type, status)
  Registration (registrationId, applicantId, unitId, date, status)
  LotteryResult (resultId, registrationId, applicantId, unitId, date, winner)

SERVICES:
  ApplicantService (implements Manageable<Applicant>)
  CondominiumService (implements Manageable<Condominium>)
  UnitService (implements Manageable<Unit>)
  RegistrationService (implements Manageable<Registration>)
  LotteryService (conductDraw, displayResults)

INTERFACE:
  Manageable<T> (add, displayAll, searchById, update)

ENUMS:
  UnitType (STUDIO, ONE_BEDROOM, TWO_BEDROOM, THREE_BEDROOM)
  UnitStatus (AVAILABLE, RESERVED, SOLD)
  RegistrationStatus (APPROVED, WON, LOST)

EXCEPTIONS:
  InvalidDataException (unchecked)
  RegistrationException (checked)

MAIN:
  Main (console UI, menu-driven, orchestrates services)
